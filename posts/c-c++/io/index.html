#include <fstream>
#include <sstream>
#include <string>

void parseFileLines(const std::string& filename) {
    std::ifstream fin(filename);
    std::string line;
    while (std::getline(fin, line)) {      // 逐行读取文件
        std::istringstream iss(line);      // 每行转为字符串流
        int x, y;
        iss >> x >> y;                     // 从字符串流解析数据
        std::cout << "Parsed: " << x << ", " << y << "\n";
    }
}
